```typescript
function handleInteraction(
  groupId: string, 
  type: InteractionType,
  gameState: GameState
): InteractionResult {
  // Verificar cooldown
  if (isInteractionBlocked(groupId, gameState)) {
    return { success: false, reason: 'COOLDOWN_ACTIVE' };
  }

  // Verificar recursos
  if (!hasRequiredResources(type, gameState)) {
    return { success: false, reason: 'INSUFFICIENT_RESOURCES' };
  }

  const interactionEffects = calculateInteractionEffects(
    groupId, 
    type, 
    gameState
  );

  // Aplicar efectos
  return applyInteractionEffects(
    groupId,
    type,
    interactionEffects,
    gameState
  );
}

function calculateInteractionEffects(
  groupId: string,
  type: InteractionType,
  gameState: GameState
): InteractionEffects {
  const group = findGroup(groupId, gameState);
  const baseEffect = INTERACTION_BASE_EFFECTS[type];
  
  // Multiplicadores por arquetipo
  const archetypeMultiplier = {
    politico: 1.2,
    sindicalista: type === 'negociar' ? 1.3 : 1.0,
    empresario: 0.9,
    comunicador: 1.1
  }[gameState.archetype];

  // Multiplicadores por influencia del grupo
  const influenceMultiplier = group.influence / 5;

  // Multiplicadores por satisfacción actual
  const satisfactionMultiplier = 
    1 + ((100 - group.satisfactionLevel) / 200);

  return {
    supportChange: baseEffect.support * 
      archetypeMultiplier * 
      influenceMultiplier * 
      satisfactionMultiplier,
    popularityChange: baseEffect.popularity * 
      archetypeMultiplier * 
      (group.popularity / 2),
    budgetChange: baseEffect.budget,
    cooldownTurns: baseEffect.cooldown
  };
}
```

3. SISTEMA DE RELACIONES
```typescript
function updateGroupRelations(
  action: GameAction,
  gameState: GameState
): GroupRelationsUpdate {
  const updates = new Map();

  // Analizar impacto en cada grupo
  gameState.interestGroups.forEach(group => {
    group.subgroups.forEach(subgroup => {
      const impact = calculateActionImpact(action, subgroup);
      
      if (impact !== 0) {
        updates.set(subgroup.id, {
          supportChange: impact,
          satisfactionChange: impact / 2,
          lastAction: action.id
        });
      }
    });
  });

  return processGroupUpdates(updates, gameState);
}

function calculateActionImpact(
  action: GameAction,
  subgroup: Subgroup
): number {
  let impact = 0;

  // Verificar intereses
  subgroup.interests.forEach(interest => {
    if (action.description.toLowerCase().includes(interest.toLowerCase())) {
      impact += 5 * subgroup.supportMultiplier;
    }
  });

  // Verificar demandas
  subgroup.demands.forEach(demand => {
    if (action.description.toLowerCase().includes(demand.toLowerCase())) {
      impact += 10 * subgroup.supportMultiplier;
    }
  });

  return impact;
}
```

```typescript
function checkGroupTensions(gameState: GameState): GroupEvent[] {
  const events = [];

  // Verificar tensiones entre grupos
  gameState.interestGroups.forEach(group => {
    group.subgroups.forEach(subgroup => {
      // Verificar satisfacción crítica
      if (subgroup.satisfactionLevel < 20) {
        events.push({
          type: 'PROTEST',
          groupId: subgroup.id,
          severity: 'HIGH',
          effects: calculateProtestEffects(subgroup)
        });
      }

      // Verificar conflictos entre grupos
      const conflicts = checkGroupConflicts(subgroup, gameState);
      events.push(...conflicts);
    });
  });

  return events;
}

function calculateProtestEffects(subgroup: Subgroup): ProtestEffects {
  return {
    popularityImpact: -5 * (subgroup.influence / 5),
    budgetImpact: -100 * (subgroup.influence / 5),
    duration: Math.ceil(subgroup.influence / 2)
  };
}
```

```typescript
function processDemands(gameState: GameState): DemandUpdate[] {
  const updates = [];

  gameState.interestGroups.forEach(group => {
    group.subgroups.forEach(subgroup => {
      // Actualizar demandas existentes
      const resolvedDemands = subgroup.demands.filter(demand =>
        isDemandResolved(demand, gameState)
      );

      // Generar nuevas demandas
      const newDemands = generateNewDemands(subgroup, gameState);

      updates.push({
        groupId: subgroup.id,
        resolvedDemands,
        newDemands,
        satisfactionChange: resolvedDemands.length * 10
      });
    });
  });

  return updates;
}

function generateNewDemands(
  subgroup: Subgroup,
  gameState: GameState
): string[] {
  const possibleDemands = DEMAND_POOL[subgroup.id];
  const currentSituation = analyzeGameState(gameState);
  
  return selectRelevantDemands(
    possibleDemands,
    currentSituation,
    subgroup
  );
}
```

```typescript
function InterestGroupPanel({ 
  group, 
  onInteraction 
}: InterestGroupPanelProps) {
  const [expandedSubgroups, setExpandedSubgroups] = useState<string[]>([]);

  const toggleSubgroup = (subgroupId: string) => {
    setExpandedSubgroups(prev =>
      prev.includes(subgroupId)
        ? prev.filter(id => id !== subgroupId)
        : [...prev, subgroupId]
    );
  };

  return (
    <div className="space-y-4">
      {group.subgroups.map(subgroup => (
        <SubgroupCard
          key={subgroup.id}
          subgroup={subgroup}
          isExpanded={expandedSubgroups.includes(subgroup.id)}
          onToggle={() => toggleSubgroup(subgroup.id)}
          onInteraction={onInteraction}
        />
      ))}
    </div>
  );
}
```

```typescript
function generateGroupFeedback(
  gameState: GameState
): GroupFeedback[] {
  return gameState.interestGroups.map(group => {
    const subgroupFeedback = group.subgroups.map(subgroup => ({
      id: subgroup.id,
      name: subgroup.name,
      status: determineStatus(subgroup),
      suggestions: generateSuggestions(subgroup, gameState),
      warnings: generateWarnings(subgroup, gameState)
    }));

    return {
      groupId: group.id,
      overallStatus: calculateGroupStatus(subgroupFeedback),
      subgroups: subgroupFeedback
    };
  });
}

function generateSuggestions(
  subgroup: Subgroup,
  gameState: GameState
): Suggestion[] {
  const suggestions = [];

  // Sugerencias basadas en satisfacción
  if (subgroup.satisfactionLevel < 40) {
    suggestions.push({
      type: 'INTERACTION',
      priority: 'HIGH',
      action: 'NEGOTIATE',
      reason: 'Low satisfaction level'
    });
  }

  // Sugerencias basadas en demandas
  subgroup.demands.forEach(demand => {
    const relevantActions = findRelevantActions(demand, gameState);
    if (relevantActions.length > 0) {
      suggestions.push({
        type: 'ACTION',
        priority: 'MEDIUM',
        actions: relevantActions,
        reason: `Addresses demand: ${demand}`
      });
    }
  });

  return suggestions;
}
```

```typescript
// Caché de cálculos costosos
const groupRelationsCache = new Map<string, {
  timestamp: number;
  relations: GroupRelations;
}>();

function getGroupRelations(
  groupId: string,
  gameState: GameState
): GroupRelations {
  const cached = groupRelationsCache.get(groupId);
  
  if (cached && 
      cached.timestamp === gameState.turn &&
      !hasRelevantStateChanges(groupId, gameState)) {
    return cached.relations;
  }

  const relations = calculateGroupRelations(groupId, gameState);
  groupRelationsCache.set(groupId, {
    timestamp: gameState.turn,
    relations
  });

  return relations;
}
```
